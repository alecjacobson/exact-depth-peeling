cmake_minimum_required(VERSION 3.20)
project(exact_depth_peeling VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CGAL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_executable(geometric_peel src/main.cpp src/exact_geometry.cpp src/camera.cpp src/triangulate_export.cpp src/visible_peel.cpp src/arrangement_oracle.cpp src/envelope_peel.cpp src/validation.cpp src/bvh_peel.cpp src/fast_peel.cpp src/fast_reference.cpp)
target_link_libraries(geometric_peel PRIVATE CGAL::CGAL nlohmann_json::nlohmann_json)
find_package(OpenMP)
set(PEEL_TEST_THREADS 1)
if(OpenMP_CXX_FOUND)
  target_link_libraries(geometric_peel PRIVATE OpenMP::OpenMP_CXX)
  set(PEEL_TEST_THREADS 4)
endif()
target_compile_options(geometric_peel PRIVATE -Wall -Wextra -Wpedantic)
add_executable(resolve_intersections src/resolve_intersections.cpp src/exact_geometry.cpp)
target_link_libraries(resolve_intersections PRIVATE CGAL::CGAL nlohmann_json::nlohmann_json)
target_compile_options(resolve_intersections PRIVATE -Wall -Wextra -Wpedantic)
option(PEEL_LAZY_EXACT "Use interval filtering with exact rational fallback" ON)
if(PEEL_LAZY_EXACT)
  target_compile_definitions(geometric_peel PRIVATE PEEL_LAZY_EXACT)
  target_compile_definitions(resolve_intersections PRIVATE PEEL_LAZY_EXACT)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME exact_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/validate.py --exe $<TARGET_FILE:geometric_peel> --threads ${PEEL_TEST_THREADS})
set_tests_properties(exact_fixtures PROPERTIES TIMEOUT 600)
add_test(NAME fast_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/validate_fast.py --exe $<TARGET_FILE:geometric_peel> --resolver $<TARGET_FILE:resolve_intersections> --threads ${PEEL_TEST_THREADS})
set_tests_properties(fast_fixtures PROPERTIES TIMEOUT 300)
