cmake_minimum_required(VERSION 3.20)
project(exact_depth_peeling VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CGAL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_executable(geometric_peel src/main.cpp src/exact_geometry.cpp src/camera.cpp src/triangulate_export.cpp src/visible_peel.cpp src/arrangement_oracle.cpp src/envelope_peel.cpp src/validation.cpp)
target_link_libraries(geometric_peel PRIVATE CGAL::CGAL nlohmann_json::nlohmann_json)
target_compile_options(geometric_peel PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME exact_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/validate.py --exe $<TARGET_FILE:geometric_peel>)
set_tests_properties(exact_fixtures PROPERTIES TIMEOUT 600)
